24class Specification(ABC): 

25 @abstractmethod 

26 def es_satisfecha_por(self, candidato: Any) -> bool: ...

28 def y(self, otra: Specification) -> Specification:

31 def o(self, otra: Specification) -> Specification:

96class DecisionCompletaSpecification(Specification): 

97 """RN-03/RN-04: la decisión cubre todos los trabajos pendientes, sin repetir, con evidencia.""" 

98 

99 def es_satisfecha_por(self, candidato: CandidatoDecision) -> bool: 

100 pendientes = {t.id for t in candidato.cotizacion.trabajos_pendientes()} 

101 autorizados, rechazados = candidato.trabajos_autorizados, candidato.trabajos_rechazados 

102 return ( 

103 bool(pendientes) 

104 and not (autorizados & rechazados) 

105 and (autorizados | rechazados) == pendientes 

106 and bool(candidato.referencia_evidencia_digital.strip()) 

107 )

127class AmpliacionSuperaUmbralSpecification(Specification): 

128 """RN-06: el acumulado supera el 20% del total autorizado; exactamente 20% no lo supera.""" 

129 

130 def es_satisfecha_por(self, candidato: CandidatoAmpliacion) -> bool: 

131 base = candidato.total_autorizado_inicial.valor 

132 if base <= 0: 

133 return candidato.ampliaciones_acumuladas.valor > 0 

134 return candidato.ampliaciones_acumuladas.valor / base > UMBRAL_AMPLIACION

144class PlazoDeRespuestaVencidoSpecification(Specification): 

145 """RN-09: 24 horas o más desde el envío de una solicitud aún sin respuesta.""" 

146 

147 def es_satisfecha_por(self, candidato: CandidatoPlazo) -> bool: 

148 orden = candidato.orden 

149 if orden.estado != EstadoOrden.EN_ESPERA_DE_AUTORIZACION: 

150 return False 

151 envios = [s.fecha_hora_envio for s in orden.solicitudes_pendientes() if s.fecha_hora_envio] 

152 return bool(envios) and candidato.ahora - min(envios) >= candidato.plazo

155class OrdenListaParaEntregaSpecification(Specification): 

156 """D1: todo lo autorizado está terminado y no queda nada pendiente.""" 

157 

158 def es_satisfecha_por(self, candidato: OrdenDeServicio) -> bool: 

159 trabajos = candidato.todos_los_trabajos() 

160 en_curso = { 

161 EstadoTrabajo.PENDIENTE_DE_AUTORIZACION, 

162 EstadoTrabajo.AUTORIZADO, 

163 EstadoTrabajo.EN_ESPERA_DE_REFACCION, 

164 EstadoTrabajo.EN_EJECUCION, 

165 } 

166 sin_ampliaciones_pendientes = all( 

167 a.estado != EstadoAmpliacion.PENDIENTE_DE_AUTORIZACION for t in trabajos for a in t.ampliaciones 

168 ) 

169 return ( 

170 any(t.estado == EstadoTrabajo.TERMINADO for t in trabajos) 

171 and not any(t.estado in en_curso for t in trabajos) 

172 and sin_ampliaciones_pendientes 

173 )
